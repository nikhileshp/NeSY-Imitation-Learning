   noop(D, 1000000, Total), !.

noop(D, TreesToUse, Total) :- // A tree-limited accessor (e.g., for tuning the number of trees to use).
   logPrior(LogPrior),
   getScore_noop_tree1(D, TreesToUse, Total1),
   Total is LogPrior + Total1,
   !.
